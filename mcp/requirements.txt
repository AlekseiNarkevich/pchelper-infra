# ВНИМАНИЕ: этот файл НЕ используется при сборке образа.
# Docker собирается по файлу requirements.lock, и туда попадает
# только рантайм - без pytest. Тестовые пакеты нужны на машине
# разработчика, а в контейнере, который работает в проде, им
# не место.
#
# ---------------------------------------------------------------------
# Версии закреплены знаком равенства (=), а не диапазонами.
# Диапазон вида >=1.2.0,<2.0 не фиксирует конкретную сборку: при
# пересборке образа в него попадёт другой код, и воспроизводимость
# пропадает. SonarCloud помечает такую установку как небезопасную.
#
# Комментарии к пакетам вынесены на отдельные строки: разборщик
# зависимостей в SonarCloud не понимает строки вида
# "пакет==версия  # пояснение" и считает версию незакреплённой.
#
# ВАЖНО: ветка 1.x зафиксирована намеренно. В mcp 2.x класс FastMCP
# переименован в MCPServer, имена декораторов и сигнатуры методов
# изменились (см. migration guide
# https://py.sdk.modelcontextprotocol.io/v2/migration/).
# Отчёт задания 5 опирается на API версии 1.x, поэтому фиксируем
# 1.x — иначе сервер падает при запуске с ModuleNotFoundError.
# ---------------------------------------------------------------------

# --- Разработка и тестирование ---

# Тестовый раннер.
# Версия 9.1.1 выбрана потому, что CVE-2025-71176 затрагивает
# pytest до 9.0.2 включительно: раннер создаёт временный каталог
# по шаблону /tmp/pytest-of-{user}, права на который зависят от
# umask. В 9.1.1 это исправлено. На 8.4.2 тесты тоже проходят, но
# SonarCloud справедливо показывает dependency risk.
pytest==9.1.1
